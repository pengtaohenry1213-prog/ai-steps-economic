
## 4. 数据格式化模块

### 4.1 `format.ts` - 单元格格式化

**文件位置**: `views/instance/edit/utils/format.ts`

**格式化规则**:

```typescript
export default function format(row, field, oldValue?) {
  const { isFixed, metricCode, unitCode, scale } = row;
  
  // 1. 单一值行的日期单元格显示为空
  if (isFixed === 0 && reg.test(field)) return '';
  
  // 2. 空值处理
  if (isEmpty(val)) return '';
  
  // 3. 字符串直接返回
  if (typeof val === 'string') return val;
  
  // 4. 绝对值处理
  if (absCodes.has(metricCode)) val = Math.abs(val);
  
  // 5. 百分比处理
  if (unitCode === 'PERCENT') return `${formatNumber(val * 100)}%`;
  
  // 6. 刻度处理
  if (scale && scale !== 0) return formatNumber(val / scale);
  
  // 7. 默认格式化
  return formatNumber(val);
}

function formatNumber(n) {
  return n.toLocaleString('zh-CN', {
    minimumFractionDigits: 1,
    maximumFractionDigits: 1,
  });
}
```
