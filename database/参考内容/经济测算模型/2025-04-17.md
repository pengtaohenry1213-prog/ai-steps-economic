```typescript
function topologicalSort(graph) {
  const visited = new Set();
  const temp = new Set();
  const order = [];
  
  function visit(node) {
    if (temp.has(node)) {
      // 检测到循环依赖
      console.warn(`检测到循环依赖: ${node}`);
      return;
    }
    
    if (!visited.has(node)) {
      temp.add(node);

      const dependencies = graph.get(node) || new Set();
      dependencies.forEach(dep => visit(dep));
      
      temp.delete(node);
      visited.add(node);
      order.push(node);
    }
  }
  
  // 遍历所有节点

    if (!visited.has(node)) {
      visit(node);
    }
  }
  
  return order.reverse();
}
```

```typescript
// 检测依赖环
function detectCycles(graph) {
  const visited = new Set();
  const recStack = new Set();
  const cycles = [];
  
  function dfs(node, path = []) {
    if (recStack.has(node)) {
      // 找到循环
      const cycle = [...path.slice(path.indexOf(node)), node];
      cycles.push(cycle);
      return true;
    }
    
    if (visited.has(node)) {
      return false;
    }
    
    visited.add(node);
    recStack.add(node);
    path.push(node);
    
    const neighbors = graph.get(node) || new Set();
    for (const neighbor of neighbors) {
      if (dfs(neighbor, path)) {
        return true;
      }
    }
    
    path.pop();
    recStack.delete(node);
    return false;
  }
  
  // 对每个节点执行DFS
  for (const node of graph.keys()) {
    if (!visited.has(node)) {
      dfs(node);
    }
  }
  
  return cycles;
}
```
