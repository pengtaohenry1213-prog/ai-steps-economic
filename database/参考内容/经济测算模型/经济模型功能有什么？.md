  cycles: Array<{
    description: string;
    path: string[];
  }>;
  hasCycle: boolean;
  inDegree: Record<string, number>;
  initialIn: Record<string, number>;
  order: string[];
  remaining: string[];
} {
  // 1. 初始化所有节点入度为0
  const inDegree: Record<string, number> = {};
  const adj: Record<string, string[]> = {};

  Object.keys(graph).forEach((key) => {
    if (!(key in inDegree)) inDegree[key] = 0;
  });
  
  // 同时把依赖节点也纳入 inDegree 的节点集合中
  Object.values(graph).forEach((deps) => {
    (deps || []).forEach((dep) => {
      if (!(dep in inDegree)) inDegree[dep] = 0;
    });
  });

  // 2. 构建反向邻接表并计算入度
  Object.entries(graph).forEach(([id, marks]) => {
    const calcMarksList = Array.isArray(marks) ? [...new Set(marks)] : [];
    calcMarksList.forEach((mark) => {
      if (!adj[mark]) adj[mark] = [];
      adj[mark].push(id);
      inDegree[id] = (inDegree[id] ?? 0) + 1;
    });
  });

  // 3. 保存初始入度值快照
  const initialIn = { ...inDegree };

  // 4. 执行 Kahn 算法 (BFS)
  const queue = Object.keys(inDegree).filter((n) => inDegree[n] === 0);
  const order: string[] = [];

  let queueIndex = 0;
  while (queueIndex < queue.length) {
    const node = queue[queueIndex++];
    order.push(node || '');

    const neighbors = adj[node || ''] || [];
    // 遍历依赖当前节点的邻居，入度-1，为0则加入队列
    for (const nbr of neighbors) {

### 🧮 算法核心：Kahn 算法 + BFS

#### `kahnTopologicalSort` 函数核心流程
这是一个完整的 **Kahn 算法 + BFS** 实现，用于拓扑排序。以下是核心流程：

##### 📌 输入输出
```typescript
// 输入: graph - 节点依赖关系图
// node -> [依赖的节点数组]，表示 node 依赖 deps 中的所有节点
graph = {
  "C10001A0319-2025": ["C10001A0322-2025", "C10001A0045-2025"],
  ...
}
// 输出: 包含排序结果、环检测信息等
```

##### 🔄 核心流程（5步）
- **第1步：初始化入度表 & 反向邻接表**
- **第2步：构建反向邻接表: dep -> [依赖 dep 的所有节点] & 计算初始入度**
	**图示说明：**
		```
		原始图 (graph):                    反向邻接表 (adj):
		  A → [B, C]        =>              B → [A]
		  B → [D]           =>              C → [A]
		  C → [D]           =>              D → [B, C]
		```
- BFS 主循环（Kahn 算法核心）**
	- 1. 入度为0的节点作为起点（无任何依赖，可以最先计算）
	- 2. 找到所有依赖当前节点的"邻居"
	- 3. 遍历邻居，入度-1，如果变为0则加入队列
	
	**执行过程图解：**
	```
	初始状态: inDegree = { A:0, B:1, C:1, D:2 }, queue = [A]
	
	第1轮: 取 A，neighbors = [B, C]
	      inDegree[B] = 1-1 = 0 → 加入 queue
	      inDegree[C] = 1-1 = 0 → 加入 queue
	      queue = [B, C], order = [A]
	
	第2轮: 取 B，neighbors = [D]
	      inDegree[D] = 2-1 = 1
	      queue = [C], order = [A, B]
	
	第3轮: 取 C，neighbors = [D]
	      inDegree[D] = 1-1 = 0 → 加入 queue
	      queue = [D], order = [A, B, C]
	
	第4轮: 取 D，neighbors = []
	      order = [A, B, C, D] ✅ 排序完成
	```

- DFS 环路径检测（如有环）**
	使用 DFS 检测环的详细路径: **找到环，提取路径 -> 递归检查依赖...**

### ✅ 核心要点总结

| 步骤 | 关键点 |
|------|--------|
| **入度计算** | 统计每个节点有多少条边指向它 |
| **BFS 队列** | 入度为0的节点可先计算 |
| **更新邻居** | 处理完一个节点后，更新其邻居的入度 |
| **环检测** | `order.length < totalNodes` 说明有环 |
| **性能优化** | 使用索引指针 `queueIndex++` 代替 `shift()` |
